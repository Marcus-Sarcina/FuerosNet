    (((PatronAcceptedRecoveryEvidence( P, S, newkey ) @ #i) ∧
      (ReliedOn( P, V, S ) @ #i)) ∧

/*

"∃ P S newkey V #i.
  (PatronAcceptedRecoveryEvidence( P, S, newkey ) @ #i) ∧
  (ReliedOn( P, V, S ) @ #i)
 ∧
  (∀ #k. (CompromisedPQ( V ) @ #k) ⇒ ⊥) ∧
  (∀ #b. (RecognisedFor( V, S, newkey ) @ #b) ⇒ ¬(#b < #i))"
